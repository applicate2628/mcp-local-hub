const selectedTests = "^TestSecretsInit_RecoveryNamespace(Boundary|RetainedDirectory|FailurePaths)$"

func main() {
	err := run()
	if retainedNamespaceStage != nil {
		// UNKNOWN remains in this original process. Neither its owner nor its
		// borrowed sinks/path pins are closed, replayed, or replaced by PID lookup.
		retainedNamespaceStage.returned = err
		fmt.Fprintln(os.Stderr, "recovery namespace fixture: UNKNOWN; original custody retained:", err)
		for {
			time.Sleep(time.Minute)
		}
	}
	if err != nil {
		fmt.Fprintln(os.Stderr, "recovery namespace fixture:", err)
		var suiteFailure *namespaceSuiteFailure
		if errors.As(err, &suiteFailure) {
			os.Exit(suiteFailure.code)
		}
		os.Exit(1)
	}
}

func run() (returned error) {
	started := time.Now()
	repoFlag := flag.String("repo", "", "absolute repository worktree")
	goFlag := flag.String("go", "", "absolute existing go.exe (no downloads)")
	pwshFlag := flag.String("pwsh", "", "absolute existing PowerShell 7 executable")
	parentFlag := flag.String("parent", os.TempDir(), "existing absolute parent for a fresh private fixture root")
	buildTimeout := flag.Duration("build-timeout", 5*time.Minute, "finite API test compilation deadline")
	testTimeout := flag.Duration("test-timeout", 3*time.Minute, "finite Go test deadline")
	testStatePathEnv := flag.Bool("test-state-path-env", false, "compile fixed tests, or require supplied tests, with test_state_path_env")
	testBinary := flag.String("test-binary", "", "exact incoming Go API image; enables one full suite without compilation")
	expectedSHA256 := flag.String("expected-sha256", "", "exact incoming API executable SHA-256")
	sourceDir := flag.String("source-dir", "", "original Go API package working directory")
	requestSHA256 := flag.String("request-sha256", "", "original full-suite issuer request SHA-256")
	requestNonce := flag.String("request-nonce", "", "original full-suite issuer request nonce")
	flag.Parse()
	wholeSuite := *testBinary != ""
	var admissionMarks []namespaceAdmissionMark
	mark := func(stage string) {
		if wholeSuite {
			admissionMarks = append(admissionMarks, namespaceAdmissionMark{Stage: stage, ElapsedNS: time.Since(started).Nanoseconds()})
		}
	}
	mark("arguments:parsed")
	if !wholeSuite && (*requestSHA256 != "" || *requestNonce != "") {
		return errors.New("issuer request binding requires a supplied API binary")
	}
	if *buildTimeout <= 0 || *testTimeout <= 0 || *buildTimeout > 10*time.Minute || *testTimeout > 10*time.Minute {
		return errors.New("invalid deadline")
	}
	if wholeSuite {
		if err := validateNamespaceSuiteArgs(flag.Args(), *testTimeout); err != nil {
			return err
		}
	} else if flag.NArg() != 0 || *expectedSHA256 != "" || *sourceDir != "" {
		return errors.New("unexpected arguments without a supplied API binary")
	}
	admission := context.Background()
	if wholeSuite {
		var cancel context.CancelFunc
		admission, cancel = context.WithDeadline(admission, started.Add(5*time.Second))
		defer cancel()
	}
	repo, err := localPath(*repoFlag)
	if err != nil {
		return err
	}
	goPath, err := localPath(*goFlag)
	if err != nil {
		return err
	}
	pwsh, err := localPath(*pwshFlag)
	if err != nil {
		return err
	}
	parent, err := localPath(*parentFlag)
	if err != nil {
		return err
	}
	var pins []windows.Handle
	defer func() {
		if retainedNamespaceStage != nil {
			retainedNamespaceStage.pathPins = pins
			retainedNamespaceStage.returned = returned
			return
		}
		for i := len(pins) - 1; i >= 0; i-- {
			returned = errors.Join(returned, windows.CloseHandle(pins[i]))
		}
	}()
	refusalBinding := namespaceRefusalBinding{
		RequestSHA256: *requestSHA256, RequestNonce: *requestNonce,
		Binary: *testBinary, ExpectedSHA256: *expectedSHA256, SourceDir: *sourceDir,
		Parent: parent, Args: append([]string(nil), flag.Args()...),
		ControllerArgv: append([]string(nil), os.Args...), TestStatePathEnv: *testStatePathEnv,
	}
	parentPinned, stageEntered := false, false
	// This runs before the pin-closing defer. It records only an original
	// admission refusal; it does not construct a contained-process settlement.
	defer func() {
		if wholeSuite {
			returned = errors.Join(returned, publishNamespacePrechildRefusal(refusalBinding, parentPinned, stageEntered, returned))
		}
	}()
	// Registered after the existing refusal publisher: preserve its exact wire
	// and original cause, adding timing DATA only to an actual pre-child error.
	defer func() {
		if wholeSuite && !stageEntered && returned != nil {
			mark("pre-child-refusal")
			if raw, err := json.Marshal(admissionMarks); err == nil {
				returned = fmt.Errorf("namespace admission marks=%s: %w", raw, returned)
			}
		}
	}()
	mark("tool-path-pins:begin")
	for index, entry := range []struct {
		path      string
		directory bool
	}{{repo, true}, {parent, true}, {goPath, false}, {pwsh, false}} {
		if err := admission.Err(); err != nil {
			return err
		}
		held, err := pinOrdinaryPath(entry.path, entry.directory)
		if err != nil {
			return err
		}
		pins = append(pins, held...)
		if index == 1 {
			parentPinned = true
		}
	}
	mark("tool-path-pins:end")
	var suite *namespaceSuiteInput
	if wholeSuite {
		suite, err = prepareNamespaceSuite(admission, repo, parent, *testBinary, *expectedSHA256, *sourceDir, *testStatePathEnv, flag.Args(), &pins, mark)
		if err != nil {
			return err
		}
	}
	diagnostics := io.Writer(os.Stdout)
	if wholeSuite {
		diagnostics = os.Stderr
	}
	mark("private-directories:begin")
	var nonce [16]byte
	if _, err := rand.Read(nonce[:]); err != nil {
		return err
	}
	root := filepath.Join(parent, "mcphub-recovery-namespace-"+hex.EncodeToString(nonce[:]))
	if wholeSuite {
		// Full mode already pinned an empty private parent. Exclusive creation
		// below preserves the same root/temp boundary with short native paths.
		root = filepath.Join(parent, "r")
	}
	tokenUser, err := windows.GetCurrentProcessToken().GetTokenUser()
	if err != nil {
		return err
	}
	sddl := "O:" + tokenUser.User.Sid.String() + "D:P(A;OICI;FA;;;" + tokenUser.User.Sid.String() + ")(A;OICI;FA;;;SY)(A;OICI;FA;;;BA)"
	sd, err := windows.SecurityDescriptorFromString(sddl)
	if err != nil {
		return err
	}
	for _, dir := range []string{root, filepath.Join(root, "temp"), filepath.Join(root, "proof"), filepath.Join(root, "bin")} {
		if err := admission.Err(); err != nil {
			return err
		}
		path, err := windows.UTF16PtrFromString(dir)
		if err != nil {
			return err
		}
		security := &windows.SecurityAttributes{Length: uint32(unsafe.Sizeof(windows.SecurityAttributes{})), SecurityDescriptor: sd}
		if err := windows.CreateDirectory(path, security); err != nil {
			return fmt.Errorf("create fresh private directory: %w", err)
		}
		held, err := pinOrdinaryPath(dir, true)
		if err != nil {
			return err
		}
		pins = append(pins, held...)
	}
	runtime.KeepAlive(sd)
	fmt.Fprintln(diagnostics, "Private evidence root:", root)
	mark("private-directories:end")

	sourceLeaves := []string{"go.mod", "go.sum", "internal/api/secrets_init_recovery_test.go", "internal/api/secrets_init_recovery_directory_windows.go", "tools/test-recovery-namespace/main_windows.go", "scripts/test-recovery-namespace/helper.ps1"}
	if wholeSuite {
		sourceLeaves = append(sourceLeaves, "internal/api/main_test.go", "internal/api/supervisor_ipc_peer_transport_windows_test.go", "internal/api/attach_client_public_cli_receiving_windows_test.go", "internal/api/supervisor_stderr_sink_test.go")
	}
	sourceHashes := make(map[string]string)
	var helper []byte
	for _, leaf := range sourceLeaves {
		if err := admission.Err(); err != nil {
			return err
		}
		mark("source-leaf:" + leaf + ":begin")
		path := filepath.Join(repo, filepath.FromSlash(leaf))
		held, err := pinOrdinaryPath(path, false)
		if err != nil {
			return err
		}
		pins = append(pins, held...)
		raw, err := os.ReadFile(path)
		if err != nil {
			return err
		}
		digest := sha256.Sum256(raw)
		sourceHashes[leaf] = hex.EncodeToString(digest[:])
		if leaf == "scripts/test-recovery-namespace/helper.ps1" {
			helper = raw
		}
		mark("source-leaf:" + leaf + ":end")
	}
	mark("helper-copy:begin")
	helperPath := filepath.Join(root, "helper.ps1")
	if err := writeFresh(helperPath, helper); err != nil {
		return err
	}
	mark("helper-copy:end")
	mark("helper-pin:begin")
	helperPins, err := pinOrdinaryPath(helperPath, false)
	if err != nil {
		return err
	}
	pins = append(pins, helperPins...)
	mark("helper-pin:end")

	binary := filepath.Join(root, "bin", "api.test.exe")
	buildArgs := []string{"test", "-mod=readonly", "-c", "-o", binary}
	if *testStatePathEnv {
		buildArgs = append(buildArgs, "-tags=test_state_path_env")
	}
	buildArgs = append(buildArgs, "./internal/api")
	tests := selectedTests
	buildArgv := append([]string{goPath}, buildArgs...)
	if wholeSuite {
		buildArgs, buildArgv, tests = nil, nil, ""
		mark("execution-copy:begin")
		if err := copyNamespaceSuiteImage(suite, binary); err != nil {
			return err
		}
		mark("execution-copy:end")
	}
	mark("inputs-proof:begin")
	if err := writeJSON(filepath.Join(root, "proof", "inputs.json"), struct {
		Repository           string            `json:"repository"`
		SelectedSourceSHA256 map[string]string `json:"selected_source_sha256"`
		Go                   string            `json:"go"`
		PowerShell           string            `json:"powershell"`
		Tests                string            `json:"tests"`
		TestStatePathEnv     bool              `json:"test_state_path_env"`
		BuildArgv            []string          `json:"build_argv"`
		Prebuilt             bool              `json:"prebuilt_full_suite,omitempty"`
		SourceDir            string            `json:"source_dir,omitempty"`
		TestArgv             []string          `json:"test_argv,omitempty"`
	}{repo, sourceHashes, goPath, pwsh, tests, *testStatePathEnv, buildArgv, wholeSuite, *sourceDir, append([]string(nil), flag.Args()...)}); err != nil {
		return err
	}
	mark("inputs-proof:end")
	mark("environment:begin")
	envValues := map[string]string{
		"GOTOOLCHAIN": "local", "GOPROXY": "off", "GOSUMDB": "off", "GOFLAGS": "", "GOWORK": "off",
		"GOOS": "windows", "GOARCH": runtime.GOARCH,
		"TEMP": filepath.Join(root, "temp"), "TMP": filepath.Join(root, "temp"), "TMPDIR": filepath.Join(root, "temp"),
		"GOTMPDIR":                                 filepath.Join(root, "temp"),
		"MCPHUB_RECOVERY_NAMESPACE_ROOT":           root,
		"MCPHUB_RECOVERY_NAMESPACE_FIXTURE_HELPER": helperPath,
		"MCPHUB_RECOVERY_NAMESPACE_FIXTURE_SHA256": sourceHashes["scripts/test-recovery-namespace/helper.ps1"],
		"MCPHUB_RECOVERY_NAMESPACE_PWSH":           pwsh,
	}
	if wholeSuite {
		// The adapter has already restored the original child Go environment.
		// Preserve that declared environment, excluding all ambient MCPHUB keys.
		for _, key := range []string{"GOTOOLCHAIN", "GOPROXY", "GOSUMDB", "GOFLAGS", "GOWORK", "GOOS", "GOARCH"} {
			delete(envValues, key)
		}
		for key, value := range suite.Environment {
			envValues[key] = value
		}
		suite.Environment = envValues
	}
	env := fixtureEnvironment(envValues)
	mark("environment:end")
	if !wholeSuite {
		build := exec.Command(goPath, buildArgs...)
		build.Dir, build.Env = repo, env
		fmt.Println("Compiling API test binary with existing local dependencies.")
		if _, err := runStage(root, "build", build, time.Now().Add(*buildTimeout)); err != nil {
			return err
		}
	}
	if err := admission.Err(); err != nil {
		return err
	}
	mark("execution-pin:begin")
	held, err := pinOrdinaryPath(binary, false)
	if err != nil {
		return err
	}
	pins = append(pins, held...)
	mark("execution-pin:end")
	mark("execution-readback:begin")
	binaryFile, err := os.Open(binary)
	if err != nil {
		return err
	}
	binaryDigest := sha256.New()
	_, hashErr := io.Copy(binaryDigest, binaryFile)
	if err := errors.Join(hashErr, binaryFile.Close()); err != nil {
		return err
	}
	mark("execution-readback:end")
	mark("binary-proof:begin")
	if err := writeJSON(filepath.Join(root, "proof", "binary.json"), map[string]string{"path": binary, "sha256": hex.EncodeToString(binaryDigest.Sum(nil))}); err != nil {
		return err
	}
	mark("binary-proof:end")
	if wholeSuite {
		if hex.EncodeToString(binaryDigest.Sum(nil)) != suite.ExpectedSHA256 {
			return errors.New("API receiver changed after admission")
		}
		suite.ExecutionBinary = binary
		suite.ExecutionSHA256 = hex.EncodeToString(binaryDigest.Sum(nil))
		if err := admission.Err(); err != nil {
			return err
		}
		mark("admission-ready")
		suite.AdmissionMarks = admissionMarks
		cmd := exec.Command(binary, suite.Args...)
		cmd.Dir, cmd.Env = suite.SourceDir, env
		workDeadline := started.Add(5*time.Second + *testTimeout)
		stageEntered = true
		receipt, runErr := runStage(root, "tests", cmd, workDeadline)
		return finishNamespaceSuite(parent, root, suite, receipt, runErr, started, workDeadline)
	}

	cmd := exec.Command(binary, "-test.run="+selectedTests, "-test.count=1", "-test.timeout="+testTimeout.String(), "-test.v")
	cmd.Dir, cmd.Env = repo, env
	fmt.Println("Running the three recovery namespace tests in a contained process.")
	receipt, runErr := runStage(root, "tests", cmd, time.Now().Add(*testTimeout+10*time.Second))
	if runErr != nil {
		return runErr
	}
	if err := verifyObservations(root, binary, receipt.RootIdentity, os.Stdout); err != nil {
		return err
	}
	fmt.Println("Namespace tests and original-process measurement bindings passed. Evidence retained:", root)
	return nil
}

type namespaceRefusalBinding struct {
	RequestSHA256    string   `json:"request_sha256"`
	RequestNonce     string   `json:"request_nonce"`
	Binary           string   `json:"binary"`
	ExpectedSHA256   string   `json:"expected_sha256"`
	SourceDir        string   `json:"source_dir"`
	Parent           string   `json:"parent"`
	Args             []string `json:"args"`
	ControllerArgv   []string `json:"controller_argv"`
	TestStatePathEnv bool     `json:"test_state_path_env"`
}

type namespacePrechildRefusal struct {
	Schema                string                  `json:"schema"`
	Binding               namespaceRefusalBinding `json:"binding"`
	ContainedStageEntered bool                    `json:"contained_stage_entered"`
	Failure               string                  `json:"failure"`
}

func publishNamespacePrechildRefusal(binding namespaceRefusalBinding, parentPinned, stageEntered bool, failure error) error {
	record, err := namespacePrechildRefusalRecord(binding, parentPinned, stageEntered, failure)
	if err != nil || record == nil {
		return err
	}
	return writeJSON(filepath.Join(binding.Parent, "namespace-suite-receipt.json"), record)
}

func namespacePrechildRefusalRecord(binding namespaceRefusalBinding, parentPinned, stageEntered bool, failure error) (*namespacePrechildRefusal, error) {
	// Legacy callers supply no issuer request and retain their existing wire.
	// Invalid or unpinned output paths never become a fallback receipt location.
	if failure == nil || !parentPinned || stageEntered || binding.RequestSHA256 == "" && binding.RequestNonce == "" {
		return nil, nil
	}
	nonce, nonceErr := hex.DecodeString(binding.RequestNonce)
	if !namespaceSHA256(binding.RequestSHA256) || nonceErr != nil || len(nonce) != 16 || strings.ToLower(binding.RequestNonce) != binding.RequestNonce {
		return nil, errors.New("invalid original issuer binding for namespace refusal")
	}
	if err := namespaceDPath(binding.Parent); err != nil {
		return nil, err
	}
	return &namespacePrechildRefusal{
		Schema: "api-namespace-prechild-refusal-v1", Binding: binding, Failure: failure.Error(),
	}, nil
}

func runStage(root, stage string, cmd *exec.Cmd, deadline time.Time) (receipt process.ContainedStreamReceiptV1, returned error) {
	var empty process.ContainedStreamReceiptV1
	if retainedNamespaceStage != nil {
		return empty, errors.New("a previous original namespace stage still retains unresolved native or IO custody")
	}
	stdout, err := os.OpenFile(filepath.Join(root, "proof", stage+".stdout.log"), os.O_CREATE|os.O_EXCL|os.O_WRONLY, 0600)
	if err != nil {
		return empty, err
	}
	closeSinks := true
	defer func() {
		if closeSinks {
			returned = errors.Join(returned, stdout.Close())
		}
	}()
	stderr, err := os.OpenFile(filepath.Join(root, "proof", stage+".stderr.log"), os.O_CREATE|os.O_EXCL|os.O_WRONLY, 0600)
	if err != nil {
		return empty, err
	}
	defer func() {
		if closeSinks {
			returned = errors.Join(returned, stderr.Close())
		}
	}()
	ctx, cancel := context.WithDeadline(context.Background(), deadline)
	defer cancel()
	process.NoConsole(cmd)
	// This single strong reference retains the original owner/error and borrowed
	// sinks if original native or IO settlement is incomplete. It is not a replacement owner.
	retainedNamespaceStage = &namespaceStageCustody{stdout: stdout, stderr: stderr}
	closeSinks = false
	receipt, runErr := process.RunContainedStreamObserved(ctx, cmd, process.ContainedStreamOptions{
		CleanupTimeout: 10 * time.Second, AbsoluteCleanupDeadline: deadline.Add(10 * time.Second),
		StdoutByteLimit: 16 << 20, StderrByteLimit: 16 << 20, Stderr: stderr,
	}, func(reader io.Reader) error { _, err := io.Copy(stdout, reader); return err })
	retainedNamespaceStage.receipt, retainedNamespaceStage.runErr = receipt, runErr
	joined := receipt.IOSettlement.Complete()
	nativeComplete := receipt.NativeSettlement != nil && receipt.NativeSettlement.Complete()
	complete := nativeComplete && joined
	if complete {
		retainedNamespaceStage = nil
		closeSinks = true
		runErr = errors.Join(runErr, stdout.Sync(), stderr.Sync())
	} else {
		runErr = errors.Join(runErr, errors.New("original namespace native or IO settlement is incomplete; original custody retained without close or replay"))
		retainedNamespaceStage.runErr = runErr
	}
	var native process.ContainedNativeSnapshot
	if receipt.NativeSettlement != nil {
		native = receipt.NativeSettlement.Snapshot()
	}
	proofErr := writeJSON(filepath.Join(root, "proof", stage+".settlement.json"), struct {
		Stage           string                          `json:"stage"`
		Root            process.ContainedRootIdentity   `json:"root"`
		Native          process.ContainedNativeSnapshot `json:"native"`
		IO              process.ContainedIOSettlement   `json:"io"`
		Complete        bool                            `json:"complete"`
		NativeComplete  bool                            `json:"native_complete"`
		IOComplete      bool                            `json:"io_complete"`
		Failure         string                          `json:"failure,omitempty"`
		WorkDeadline    time.Time                       `json:"work_deadline"`
		CleanupDeadline time.Time                       `json:"absolute_cleanup_deadline"`
		RootWaitKnown   bool                            `json:"root_wait_known"`
		RootExitCode    int                             `json:"root_exit_code"`
		CustodyRetained bool                            `json:"custody_retained"`
	}{stage, receipt.RootIdentity, native, receipt.IOSettlement, complete, nativeComplete, joined, errorText(runErr), deadline, deadline.Add(10 * time.Second), receipt.RootWaitKnown, receipt.RootExitCode, retainedNamespaceStage != nil})
	if runErr != nil || proofErr != nil {
		return receipt, errors.Join(runErr, proofErr)
	}
	if !complete || !receipt.RootWaitKnown || receipt.RootExitCode != 0 {
		return receipt, errors.New(stage + ": positive process/IO settlement is incomplete")
	}
	return receipt, nil
}

func verifyObservations(root, binary string, identity process.ContainedRootIdentity, diagnostics io.Writer) error {
	if !identity.Known || identity.PID <= 0 || identity.NativeBirthFILETIME == 0 {
		return errors.New("original actor identity was not observed")
	}
	paths, err := filepath.Glob(filepath.Join(root, "proof", "handles-*.json"))
	if err != nil {
		return err
	}
	if len(paths) != 4 {
		return fmt.Errorf("expected four real handle observations from two test boundaries, received %d", len(paths))
	}
	for _, path := range paths {
		raw, err := os.ReadFile(path)
		if err != nil {
			return err
		}
		var observed struct {
			ActorPID uint32 `json:"ActorPid"`
			Birth    uint64 `json:"CreationTimeFILETIME"`
			Image    string `json:"ExecutablePath"`
			Count    uint32 `json:"HandleCount"`
		}
		if err := json.Unmarshal(raw, &observed); err != nil {
			return err
		}
		if observed.ActorPID != uint32(identity.PID) || observed.Birth != identity.NativeBirthFILETIME || !strings.EqualFold(filepath.Clean(observed.Image), binary) || observed.Count == 0 {
			return errors.New("handle measurement does not match the original contained actor")
		}
	}
	raw, err := os.ReadFile(filepath.Join(root, "proof", "wrong-owner-attempt.json"))
	if err != nil {
		return err
	}
	var owner struct {
		Assigned                                bool
		Failure                                 string
		Attempts                                int
		OwnerBefore, OwnerAfter, RequestedOwner string
	}
	if err := json.Unmarshal(raw, &owner); err != nil {
		return err
	}
	if owner.Attempts != 1 || owner.OwnerBefore == "" || owner.RequestedOwner != "S-1-5-32-545" {
		return errors.New("wrong-owner attempt evidence is incomplete")
	}
	if owner.Assigned {
		if owner.Failure != "" || owner.OwnerAfter != owner.RequestedOwner {
			return errors.New("wrong-owner success did not read back")
		}
		fmt.Fprintln(diagnostics, "Wrong-owner rejection exercised after one successful private owner assignment.")
	} else {
		if owner.Failure == "" || owner.OwnerAfter != owner.OwnerBefore {
			return errors.New("wrong-owner refusal evidence is inconsistent")
		}
		fmt.Fprintln(diagnostics, "UNVERIFIED actual wrong-owner receiving: ordinary-token assignment was refused; see private proof.")
	}
	return nil
}

func localPath(path string) (string, error) {
	clean := filepath.Clean(path)
	if !filepath.IsAbs(path) || len(clean) < 3 || clean[1] != ':' || clean[2] != '\\' || !((clean[0] >= 'A' && clean[0] <= 'Z') || (clean[0] >= 'a' && clean[0] <= 'z')) {
		return "", errors.New("an absolute local drive path is required")
	}
	for _, part := range strings.Split(clean[3:], `\`) {
		if strings.HasSuffix(part, ".") || strings.HasSuffix(part, " ") || strings.ContainsAny(part, `:"<>|?*`) {
			return "", errors.New("ambiguous path component refused")
		}
	}
	return clean, nil
}

// Retain every ordinary ancestor without write/delete sharing, preventing both
// replacement and in-place reparse changes. Source bytes stay bound as well.
func pinOrdinaryPath(path string, directory bool) (held []windows.Handle, returned error) {
	defer func() {
		if returned != nil {
			for _, handle := range held {
				_ = windows.CloseHandle(handle)
			}
			held = nil
		}
	}()
	parts := []string{filepath.VolumeName(path) + `\`}
	for _, part := range strings.Split(strings.TrimPrefix(path, parts[0]), `\`) {
		if part != "" {
			parts = append(parts, filepath.Join(parts[len(parts)-1], part))
		}
	}
	for i, part := range parts {
		name, err := windows.UTF16PtrFromString(part)
		if err != nil {
			return held, err
		}
		leaf := i == len(parts)-1
		access, share := uint32(windows.READ_CONTROL|windows.FILE_READ_ATTRIBUTES), uint32(windows.FILE_SHARE_READ)
		if leaf && !directory {
			access |= windows.GENERIC_READ
			share = windows.FILE_SHARE_READ
		}
		handle, err := windows.CreateFile(name, access, share, nil, windows.OPEN_EXISTING, windows.FILE_FLAG_OPEN_REPARSE_POINT|windows.FILE_FLAG_BACKUP_SEMANTICS, 0)
		if err != nil {
			return held, err
		}
		held = append(held, handle)
		var info windows.ByHandleFileInformation
		if err := windows.GetFileInformationByHandle(handle, &info); err != nil {
			return held, err
		}
		if info.FileAttributes&windows.FILE_ATTRIBUTE_REPARSE_POINT != 0 || (!leaf || directory) != (info.FileAttributes&windows.FILE_ATTRIBUTE_DIRECTORY != 0) {
			return held, errors.New("reparse or unexpected file type in fixture path")
		}
	}
	return held, nil
}

func fixtureEnvironment(values map[string]string) []string {
	env := make([]string, 0, len(os.Environ())+len(values))
	for _, entry := range os.Environ() {
		key, _, _ := strings.Cut(entry, "=")
		// API TestMain has child-process sentinels before its sandbox setup.
		// A stale inherited sentinel must never select an external state root.
		if strings.HasPrefix(strings.ToUpper(key), "MCPHUB_") {
			continue
		}
		if _, replaced := values[strings.ToUpper(key)]; !replaced {
			env = append(env, entry)
		}
	}
	keys := make([]string, 0, len(values))
	for key := range values {
		keys = append(keys, key)
	}
	sort.Strings(keys)
	for _, key := range keys {
		env = append(env, key+"="+values[key])
	}
	return env
}

func writeFresh(path string, raw []byte) (err error) {
	file, err := os.OpenFile(path, os.O_CREATE|os.O_EXCL|os.O_WRONLY, 0600)
	if err != nil {
		return err
	}
	defer func() { err = errors.Join(err, file.Close()) }()
	if _, err = file.Write(raw); err != nil {
		return err
	}
	return file.Sync()
}

func writeJSON(path string, value any) error {
	raw, err := json.MarshalIndent(value, "", "  ")
	if err != nil {
		return err
	}
	return writeFresh(path, append(raw, '\n'))
}

func errorText(err error) string {
	if err != nil {
		return err.Error()
	}
	return ""
}
