def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()

def new_json(path, value):
    with Path(path).open("x", encoding="utf-8", newline="\n") as output:
        json.dump(value, output, indent=2)
        output.write("\n")

def require(condition, message):
    if not condition:
        raise RuntimeError(message)

def verify_inputs(expected_runner_sha256):
    require(REVIEW_COMPLETE is True, "disabled: root review, fresh tool pin and new scope required")
    require(os.name == "nt" and ROOT.is_relative_to(ALLOWED_PARENT), "fresh D task scope required")
    require(Path.cwd() == ROOT and sys.flags.isolated and sys.flags.no_site and sys.flags.dont_write_bytecode,
            "exact isolated interpreter flags and runner directory required")
    require(sha(__file__) == expected_runner_sha256, "reviewed qualification runner differs")
    require(sha(sys.executable) == PYTHON_SHA256, "installed Python differs")
    require(len(GO_SHA256) == 64 and sha(GO) == GO_SHA256, "read-only installed Go pin required")
    require(sha(JOB_MODULE) == JOB_SHA256, "installed original Windows owner differs")
    for name, expected in INPUTS.items():
        require(len(expected) == 64 and sha(ROOT / name) == expected, "candidate input differs: " + name)

    sys.modules[spec.name] = fixtures
    spec.loader.exec_module(fixtures)
    fixtures.OUT = OUT / "cases"
    fixtures.NATIVE = True

    # It executes actual candidate guard and broker methods, and extracts only
    # the pinned RootInvocation/retention methods to avoid importing full-run data.
    try:
        return fixtures.main()

        fixtures.retain_unsettled_fixtures()
