def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()

def new_json(path, value):
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open("x", encoding="utf-8", newline="\n") as f:
        json.dump(value, f, indent=2)
        f.write("\n")

def ordinary(path, directory=None):
    path = Path(os.path.abspath(path))
    for p in list(reversed(path.parents)) + [path]:
        info = p.lstat()
        if getattr(info, "st_file_attributes", 0) & REPARSE or stat.S_ISLNK(info.st_mode):
            raise RuntimeError("reparse path refused: " + str(p))
    if directory is True and not path.is_dir():
        raise RuntimeError("directory required: " + str(path))
    if directory is False and not path.is_file():
        raise RuntimeError("ordinary file required: " + str(path))
    return path

def tree(root):
    root = ordinary(root, True)
    result = {}
    for p in sorted(root.rglob("*")):
        ordinary(p)
        if p.is_file():
            result[p.relative_to(root).as_posix()] = {"bytes":p.stat().st_size, "sha256":sha(p)}
    return result

def copy_file(src, dst):
    src = ordinary(src, False)
    dst = Path(dst)
    dst.parent.mkdir(parents=True, exist_ok=True)
    with src.open("rb") as reader, dst.open("xb") as writer:
        shutil.copyfileobj(reader, writer)
    if sha(src) != sha(dst):
        raise RuntimeError("copy hash mismatch: " + str(src))
    return sha(dst)

def copy_tree(src, dst):
    before = tree(src)
    dst.mkdir(parents=True, exist_ok=False)
    for rel in before:
        copy_file(src / Path(rel), dst / Path(rel))
    if tree(src) != before or tree(dst) != before:
        raise RuntimeError("tree changed during copy")
    return before

                or state.get("state") != "complete" or state.get("direct_exit_known") is not True
                or Path(row.get("stdout", "")).name != state.get("name", "") + ".stdout.txt"
                or any(not isinstance(row.get(stream + "_sha256"), str)
                       or not re.fullmatch(r"[0-9a-f]{64}", row[stream + "_sha256"])
                       for stream in ("stdout", "stderr"))):
            raise RuntimeError("prior original task command/capture metadata is incomplete")
    if prior == FOCUSED_RUN:
        if (original.get("schema") != "private-serena-native-focused-run-v1"
                or original.get("full_gate_passed") is not False
                or original.get("literal_pre_push_commands_passed") is not False
                or received.get("full_gate_passed") is not False
                or received.get("literal_pre_push_commands_passed") is not False):
            raise RuntimeError("focused native gate must not claim the full regression gates")
    dependency_files = {name: snapshot["files"].get(name) for name in ("go.mod", "go.sum")}
    if any(not isinstance(digest, str) or not re.fullmatch(r"[0-9a-f]{64}", digest)
           or sha(ordinary(REPO / name, False)) != digest for name, digest in dependency_files.items()):
        raise RuntimeError("current dependency definition differs from the exact task predecessor")
    return {"path": str(path), "sha256": hashlib.sha256(raw).hexdigest(), "record": handoff}

def receive_cache_bootstrap(run, expected):
    """Receive the existing bootstrap for its same-run compiler participant or successor."""
    global CACHE_BOOTSTRAP_SOURCE
    run = Path(run)
    path = run / "cache-bootstrap.json"
    if (run != BOOTSTRAP_RUN or not isinstance(expected, dict)
            or expected.get("path") != str(path)
            or not isinstance(expected.get("sha256"), str)
            or not re.fullmatch(r"[0-9a-f]{64}", expected["sha256"])):
        raise RuntimeError("one pinned original private-default bootstrap is required")
    raw = ordinary(path, False).read_bytes()
    record = json.loads(raw)
    snapshot, dependencies = receive_bootstrap_source()
    caches = {"GOMODCACHE": str(SHARED_MODULE_CACHE), "GOCACHE": str(BUILD_CACHE)}
    if (hashlib.sha256(raw).hexdigest() != expected["sha256"] or record != expected.get("record")
            or record.get("schema") != "go-cache-bootstrap-v1"
            or record.get("consumer_run") != str(run) or record.get("consumer_source_commit") != CACHE_COMMIT
            or record.get("prior_cache_consumer") is not None
            or record.get("source_admission") != SOURCE_ADMISSION or record.get("dependency_files") != dependencies
            or record.get("shared_cache_paths") != caches or set(record.get("created_caches", {})) != set(caches)):
        raise RuntimeError("original private-default cache bootstrap control differs")
    for key, value in record["created_caches"].items():
        cache = Path(caches[key])
        info = ordinary(cache, True).stat()
        if (cache.parent != run or value.get("path") != str(cache)
                or value.get("created_exclusively") is not True or value.get("empty_at_creation") is not True
                or type(value.get("inode")) is not int or value["inode"] <= 0
                or info.st_ino != value["inode"] or info.st_dev != value.get("device")):
            raise RuntimeError("original private-default cache object identity differs")
    CACHE_BOOTSTRAP_SOURCE = snapshot
    return {"path": str(path), "sha256": hashlib.sha256(raw).hexdigest(), "record": record}


def create_cache_bootstrap(run):
    """Create the fixed first consumer's two caches exclusively, without a prior handoff."""
    global CACHE_BOOTSTRAP_SOURCE
    run = Path(run)
    if run != BOOTSTRAP_RUN or CACHE_BOOTSTRAP is not None or CACHE_HANDOFF is not None:
        raise RuntimeError("cache bootstrap belongs only to the one fresh first consumer")
    snapshot, dependencies = receive_bootstrap_source()
    ordinary(run, True)
    caches = {"GOMODCACHE": SHARED_MODULE_CACHE, "GOCACHE": BUILD_CACHE}
    created = {}
    for key, path in caches.items():
        if path.parent != run:
            raise RuntimeError("fresh cache must be a direct child of its original private run")
        path.mkdir(exist_ok=False)
        info = ordinary(path, True).stat()
        if type(info.st_ino) is not int or info.st_ino <= 0 or any(path.iterdir()):
            raise RuntimeError("new cache identity/empty creation was not observed")
        created[key] = {"path": str(path), "device": info.st_dev, "inode": info.st_ino,
                        "created_exclusively": True, "empty_at_creation": True}
    record = {"schema": "go-cache-bootstrap-v1", "consumer_run": str(run),
              "consumer_source_commit": CACHE_COMMIT, "prior_cache_consumer": None,
              "source_admission": SOURCE_ADMISSION, "dependency_files": dependencies,
              "shared_cache_paths": {key: str(path) for key, path in caches.items()},
              "created_caches": created,
              "created_utc": time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime())}
    path = run / "cache-bootstrap.json"
    new_json(path, record)
    CACHE_BOOTSTRAP_SOURCE = snapshot
    return {"path": str(path), "sha256": sha(path), "record": record}

HEADLESS_BYTES = (b"projects: []\n"
                  b"gui_log_window: false\n"
                  b"web_dashboard: false\n"
                  b"web_dashboard_open_on_launch: false\n")
HEADLESS_SHA = "96f52df38eddbfd3bef53d29ca5f445099a2c63e292b06be8c63616172ec1484"

def headless_config_snapshot(root, raw):
    """Pure DATA for our own generated YAML; matching the headless10 predicate."""
    if type(raw) is not bytes or len(raw) > 1 << 20:
        raise RuntimeError("headless configuration exceeds its fixed bound")
    text = raw.decode("utf-8").replace("\r\n", "\n")
    for key in HEADLESS_SETTINGS:
        if (len(re.findall(r"^" + key + r"[ \t]*:", text, re.MULTILINE)) != 1
                or len(re.findall(r"^" + key + r":[ \t]*false[ \t]*(?:#.*)?$", text, re.MULTILINE)) != 1):
            raise RuntimeError("private headless setting changed: " + key)
    if len(re.findall(r"^projects[ \t]*:", text, re.MULTILINE)) != 1:
        raise RuntimeError("required Serena projects field is missing or duplicated")
    return {"path": str(Path(root) / "profile/serena_config.yml"), "bytes": len(raw),
            "sha256": hashlib.sha256(raw).hexdigest(), "settings": dict(HEADLESS_SETTINGS),
            "projects_are_mutable_output": True}


def prepare_headless_environment(root, env):
    """Exclusive immutable seed and mutable private copy, before any test launch."""
    root = ordinary(root, True)
    if root.parent != RUNS or hashlib.sha256(HEADLESS_BYTES).hexdigest() != HEADLESS_SHA:
        raise RuntimeError("fixed private headless scope and seed required")
    for path in (root / "headless-config-input.yml", root / "profile/serena_config.yml"):
        ordinary(path.parent, True)
        with path.open("xb") as stream:
            stream.write(HEADLESS_BYTES)
            stream.flush()
            os.fsync(stream.fileno())
        if sha(ordinary(path, False)) != HEADLESS_SHA:
            raise RuntimeError("new private headless configuration differs")
    env["SERENA_HOME"] = str(root / "profile")


def receive_headless_before(root, env):
    """Called immediately before the original full-suite owner is acquired."""
    root = ordinary(root, True)
    seed, config = root / "headless-config-input.yml", root / "profile/serena_config.yml"
    if env.get("SERENA_HOME") != str(root / "profile") or root.parent != RUNS:
        raise RuntimeError("actual full-suite headless environment escaped its private scope")
    for path in (seed, config):
        with ordinary(path, False).open("rb") as stream:
            raw = stream.read((1 << 20) + 1)
        if raw != HEADLESS_BYTES or hashlib.sha256(raw).hexdigest() != HEADLESS_SHA:
            raise RuntimeError("fixed headless seed/config changed before suite creation")
    return {"schema": "serena-private-headless-config-v1",
        "seed": {"path": str(seed), "sha256": HEADLESS_SHA, "bytes": len(HEADLESS_BYTES)},
        "mutable_config": str(config), "settings": dict(HEADLESS_SETTINGS), "initial_projects_empty": True,
        "before": headless_config_snapshot(root, HEADLESS_BYTES), "before_launch_verified": True,
        "scope": "supported private headless configuration; default GUI behavior is not qualified"}


def receive_headless_after(root, env, record):
    """Read only after original native/IO closure; projects are mutable output."""
    root = ordinary(root, True)
    seed, config = root / "headless-config-input.yml", root / "profile/serena_config.yml"
    if (root.parent != RUNS or env.get("SERENA_HOME") != str(root / "profile")
            or record.get("schema") != "serena-private-headless-config-v1"
            or record.get("seed") != {"path": str(seed), "sha256": HEADLESS_SHA, "bytes": len(HEADLESS_BYTES)}
            or record.get("mutable_config") != str(config) or record.get("settings") != HEADLESS_SETTINGS
            or record.get("initial_projects_empty") is not True or record.get("before_launch_verified") is not True
            or record.get("before") != headless_config_snapshot(root, HEADLESS_BYTES)
            or record.get("scope") != "supported private headless configuration; default GUI behavior is not qualified"):
        raise RuntimeError("original full-suite headless configuration binding differs")
    with ordinary(seed, False).open("rb") as stream:
        seed_raw = stream.read((1 << 20) + 1)
    if seed_raw != HEADLESS_BYTES or hashlib.sha256(seed_raw).hexdigest() != HEADLESS_SHA:
        raise RuntimeError("immutable headless input changed during the full suite")
    with ordinary(config, False).open("rb") as stream:
        raw = stream.read((1 << 20) + 1)
    return headless_config_snapshot(root, raw)

    def save(self):
        (self.root/"receipt.json").write_text(json.dumps(self.receipt, indent=2)+"\n", encoding="utf-8")
    def command(self, argv, name, cwd=REPO, timeout=600, check=True, stdin_path=None):
        row = {"argv":[str(x) for x in argv], "cwd":str(cwd), "stdout":str(self.root/(name+".stdout.txt")), "stderr":str(self.root/(name+".stderr.txt"))}
        self.receipt["commands"].append(row)
        self.save()
        start = time.monotonic()
        try:
            with Path(row["stdout"]).open("xb") as out, Path(row["stderr"]).open("xb") as err:
                stdin = Path(stdin_path).open("rb") if stdin_path else subprocess.DEVNULL
                try:
                    done = subprocess.run(row["argv"], cwd=str(cwd), env=self.env, stdin=stdin, stdout=out, stderr=err,
                        timeout=timeout, check=False, creationflags=subprocess.CREATE_NO_WINDOW)
                finally:
                    if stdin_path:
                        stdin.close()
            row["exit_code"] = done.returncode
        finally:
            row["seconds"] = round(time.monotonic()-start, 3)
            for stream in ("stdout","stderr"):
                if Path(row[stream]).exists():
                    row[stream+"_sha256"] = sha(row[stream])
            self.save()
        if check and done.returncode:
            raise RuntimeError(name+" exited "+str(done.returncode)+"; see "+row["stderr"])
        return Path(row["stdout"]).read_text(encoding="utf-8-sig", errors="strict")
    def validator(self, flags, name):
        return json.loads(self.command([GO, "run", "./tools/windows-product-validate", *flags], name, timeout=180))
    def retain_failed_full_workers(self, failure_published):
        """Keep this actual failed owner; receive closure without changing its gate."""
        pending_exception = sys.exc_info()[0] is not None
        original_receipt = self.root / "receipt.json"
        metadata_error = None
        try:
            original_digest = sha(original_receipt)
        except BaseException as error:
            original_digest = None
            metadata_error = error
        original_states = self.quiescence_snapshot()
        while True:
            try:
                proof = self.receive_failed_worker_closure(time.monotonic() + 30)
                valid = (proof.get("schema") == "failed-original-worker-physical-closure-v1"
                    and proof.get("original_command_states_unchanged") is True
                    and proof.get("capture_access_issued") is False
                    and proof.get("new_commands_permitted") is False
                    and proof.get("qualification_passed") is False
                    and self.quiescence_snapshot() == original_states)
                if (valid and proof.get("status") == "CLOSED" and proof.get("commands")
                        and all(row.get("state") == "CLOSED" for row in proof["commands"])):
                    break
            except BaseException:
                pass
            # The actual Run, native owner and original workers remain retained.
            # This wait grants no qualification time and starts no new work.
            try:
                time.sleep(0.05)
            except BaseException:
                continue
        try:
            new_json(self.root / "failed-original-physical-closure.json", {
                "schema": "literal-full-failed-original-physical-closure-v1", "status": "CLOSED",
                "original_receipt": {"path": str(original_receipt), "sha256": original_digest},
                "receipt_digest_scope": "actual observed on-disk metadata only",
                "original_failure_publication_completed": failure_published,
                "intended_failure_receipt_bound": failure_published and original_digest is not None,
                "original_gate_status": "FAIL", "original_gate_reclassified": False,
                "original_command_states": original_states, "commands_quiescent": False,
                "physical_receiving": proof, "capture_access_issued": False,
                "new_commands_permitted": False, "source_or_cache_access_issued": False,
                "qualification_deadlines_extended": False})
        except BaseException:
            # Preserve an earlier save/print failure after actual physical closure.
            if not pending_exception:
                raise
        if metadata_error is not None and not pending_exception:
            raise metadata_error

    def finish(self, status, error=None):
        owned_failure = (Path(self.root) == FULL_RUN and any(
            item["state"] != "complete" and any(custody.get("run") is self and custody["row"] is item["row"]
                for custody in _guard.RETAINED_COMMANDS)
            for item in self._command_receiving.values()))
        if owned_failure:
            status = "FAIL"
            self.receipt["commands_quiescent"] = False
            self.receipt["command_quiescence"] = self.quiescence_snapshot()
        self.receipt["status"] = status
        self.receipt["finished_utc"] = time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime())
        if error is not None:
            self.receipt["error"] = type(error).__name__+": "+str(error)
        published = False
        try:
            self.save()
            published = True
            print(json.dumps({"status":status,"run":str(self.root),"receipt":str(self.root/"receipt.json")}), flush=True)
        finally:
            if owned_failure:
                self.retain_failed_full_workers(published)

def complete_product_cleanup(run, artifact):
    # commands_quiescent inspects the still-retained original guard owners.
    # No filesystem observation is allowed on the unresolved path.
    if not run.commands_quiescent():
        raise RuntimeError("original Product command/native/IO custody is unresolved")
    rows = [row for row in run.receipt["commands"] if Path(row["stdout"]).name == "actual-builder.stdout.txt"]
    if len(rows) != 1:
        raise RuntimeError("one original actual-builder command required before disposition")
    row = rows[0]
    if (row.get("quiescence_state") != "complete" or row.get("original_owner_retained") is not True
            or row.get("original_readers_joined") is not True or row.get("captures_measured") is not True
            or row.get("completion_receiving", {}).get("lifecycle_state") != "CLOSED"
            or row.get("native", {}).get("active_zero") is not True
            or row.get("native", {}).get("handles_closed") is not True
            or row.get("native", {}).get("direct_reaped") is not True):
        raise RuntimeError("original actual-builder closure differs")
    path = run.root / "deferred-cleanup.json"
    record = json.loads(ordinary(path, False).read_bytes())
    pid = record.get("builder_pid")
    if (record.get("schema") != "product-deferred-owned-cleanup-v1" or type(pid) is not int
            or pid != row.get("original_pid") or record.get("source") != str(REPO / "build.ps1")
            or record.get("artifact") != str(artifact) or not isinstance(record.get("targets"), list)):
        raise RuntimeError("closed builder cleanup-plan attribution differs")
    targets, seen, roles = [], set(), []
    for target in record["targets"]:
        if not isinstance(target, dict) or set(target) != {"path", "directory", "role"} or type(target["directory"]) is not bool:
            raise RuntimeError("closed builder cleanup target shape differs")
        source, role = Path(target["path"]), target["role"]
        allowed = ((role == "product-resource" and source == PRODUCT_SYSO and target["directory"] is False)
            or (role == "resource-scratch" and source.parent == REPO / ".scratch" and target["directory"] is True
                and re.fullmatch(r"windows-versioninfo-" + str(pid) + r"-[0-9a-f]{32}", source.name))
            or (role == "runtime-scratch" and source.parent == artifact and target["directory"] is True
                and re.fullmatch(r"mcphub-rust-owner-" + str(pid) + r"-[0-9a-f]{32}", source.name))
            or (role == "incomplete-runtime" and target["directory"] is False and row.get("exit_code") != 0
                and source in (artifact / "mcphub-product/mcphub-owner.dll", artifact / "mcphub-product/mcphub-owner-identity.json")))
        if not allowed or source in seen:
            raise RuntimeError("unowned or repeated deferred cleanup path")
        seen.add(source)
        roles.append(role)
        targets.append((source, target["directory"], role))
    if row.get("exit_code") == 0 and sorted(roles) != ["product-resource", "resource-scratch", "runtime-scratch"]:
        raise RuntimeError("successful Product cleanup inventory must contain its exact three targets")
    # Validate all metadata before observing or disposing any target.
    records = []
    for source, directory, role in targets:
        if os.path.lexists(source):
            ordinary(source, directory)
            if role == "resource-scratch":
                if any(child.name != "product.json" or not child.is_file() for child in source.iterdir()):
                    raise RuntimeError("resource scratch contains an unowned leaf")
            records.append(cleanup_move(run.root, source, pid, owner_run=run))
        elif row.get("exit_code") == 0:
            raise RuntimeError("successful builder did not retain its deferred cleanup target")
    run.receipt["deferred_cleanup"] = {"path": str(path), "sha256": sha(path), "original_pid": pid,
        "native_and_io_closed_before_disposition": True, "records": records}
    run.save()


def product_source_snapshot(run, suffix):
    snapshot = source_snapshot(run, suffix)
    status = run.command([GIT, "--no-optional-locks", "-c", "core.fsmonitor=false", "-c", "core.untrackedCache=false",
                          "-C", REPO, "status", "--porcelain=v1", "--untracked-files=all", "-z"],
                         "clean-" + suffix, timeout=30)
    if status or not run.commands_quiescent() or len(snapshot["files"]) != SOURCE_COUNT:
        raise RuntimeError("Product source snapshot is not fully closed and clean")
    # The owned artifact tree is disposed before the final observation. Detect
    # even ignored extra files; .git is the sole excluded control directory.
    observed = set()
    for folder, directories, names in os.walk(REPO, topdown=True, followlinks=False):
        directory = Path(folder)
        if directory == REPO:
            directories[:] = [name for name in directories if name != ".git"]
        for name in directories:
            ordinary(directory / name, True)
        for name in names:
            path = ordinary(directory / name, False)
            observed.add(path.relative_to(REPO).as_posix())
    if observed != set(snapshot["files"]):
        raise RuntimeError("complete Product source inventory contains unadmitted files")
    controls = SEED_SOURCE_RECEIVED["original"]["git_controls"]
    if any(sha(ordinary(REPO / ".git" / name, False)) != digest for name, digest in controls.items()):
        raise RuntimeError("Product changed issued source Git controls")
    return snapshot


def bundle_admission(run, bundle, plan, name, native_sha):

                "product_sha256":sha(product),"identity_sha256":sha(identity)}

def resource_tool_proxy(root):
    # Only the two complete tool tuples in the already accepted archive import.
    prepared = receive_module_proxy()
    proxy = root / "resource-tool-proxy"
    proxy.mkdir(exist_ok=False)
    sources, versions = {}, {}
    for module, version in (("github.com/josephspurrier/goversioninfo", "v1.5.0"),
                            ("github.com/akavel/rsrc", "v0.10.2")):
        relbase = Path(module) / "@v" / version
        for suffix in (".info", ".mod", ".zip"):
            relative = Path(str(relbase) + suffix)
            source = ordinary(MODULE_PROXY / relative, False)
            expected = prepared["manifest"].get(relative.as_posix())
            if expected is None or source.stat().st_size != expected["bytes"] or sha(source) != expected["sha256"]:
                raise RuntimeError("imported resource-tool input differs from pinned receipt: " + str(source))
            if suffix == ".info" and json.loads(source.read_text(encoding="utf-8")).get("Version") != version:
                raise RuntimeError("imported resource-tool version metadata disagrees")
            copy_file(source, proxy / relative)
            sources[str(source)] = expected["sha256"]
        index = proxy / Path(module) / "@v" / "list"
        with index.open("x", encoding="ascii", newline="\n") as f:
            f.write(version + "\n")

              "scope": "Accepted task-local archive bytes plus a per-run @v/list; no donor cache access or network"}

    return proxy, result

def pinned_json(path, expected, limit=16 << 20):
    with ordinary(path, False).open("rb") as reader:
        raw = reader.read(limit + 1)
    if len(raw) > limit or not re.fullmatch(r"[0-9a-f]{64}", expected) or hashlib.sha256(raw).hexdigest() != expected:
        raise RuntimeError("original bounded control changed: " + str(path))
    return json.loads(raw)

            or original.get("source_before") != native_snapshot_source
            or any(original.get(key) is not True for key in
                   ("source_unchanged", "offline_cache_unchanged", "inputs_unchanged", "commands_quiescent"))
            or len(native_sources) != 64 or received.get("native_sources") != native_sources
            or original.get("native_sources_before") != native_sources or original.get("native_sources_after") != native_sources
            or any(native_snapshot_source["files"].get(name) != digest
                   or snapshot["files"].get(name) != digest for name, digest in native_sources.items())):
        raise RuntimeError("original H6 Rust compilation or exact unchanged current native map differs")
    rows = original.get("commands", [])
    quiet = original.get("command_quiescence", {})
    states = quiet.get("commands", [])
    if (not rows or len(rows) != len(states) or quiet.get("complete") is not True
            or received.get("captures_verified") != 2 * len(rows)):
        raise RuntimeError("received H6 Rust original command/IO completion differs")
    for row, state in zip(rows, states):
        if (type(row.get("exit_code")) is not int or row["exit_code"] != 0
                or row.get("quiescence_state") != "complete" or row.get("captures_measured") is not True
                or state.get("state") != "complete" or state.get("direct_exit_known") is not True or state.get("reason")
                or Path(row.get("stdout", "")).name != state.get("name", "") + ".stdout.txt"
                or any(not isinstance(row.get(side + "_sha256"), str)
                       or not re.fullmatch(r"[0-9a-f]{64}", row[side + "_sha256"]) for side in ("stdout", "stderr"))):
            raise RuntimeError("received H6 Rust original command/capture metadata is incomplete")
    snapshot_record = binding["source_snapshot"]
    snapshot_path = Path(snapshot_record["path"])
    if (snapshot_path.parent != native_run or received.get("source_snapshot") != snapshot_record
            or pinned_json(snapshot_path, snapshot_record["sha256"]) != native_snapshot_source):
        raise RuntimeError("original H6 native compiler source snapshot differs")
    inputs = original.get("input_files", {})
    if not inputs or received.get("inputs_verified") != len(inputs):
        raise RuntimeError("original H6 native compiler input inventory differs")
    for path, digest in inputs.items():
        if (not isinstance(digest, str) or not re.fullmatch(r"[0-9a-f]{64}", digest)
                or sha(ordinary(path, False)) != digest):
            raise RuntimeError("original H6 native compiler input changed: " + path)
    # These are build artifacts, first read after the received original controls.
    for key in ("library", "member_child"):
        artifact = binding[key]
        path = Path(artifact["path"])
        if (received.get(key) != artifact or original.get(key) != artifact
                or not path.is_relative_to(native_run / "targets")
                or type(artifact.get("bytes")) is not int or artifact["bytes"] <= 0
                or ordinary(path, False).stat().st_size != artifact["bytes"]
                or sha(path) != artifact["sha256"]):
            raise RuntimeError("retained H6 native output changed: " + key)
    current_native_inputs = {}
    for name, digest in native_sources.items():
        path = ordinary(REPO / name, False)
        if sha(path) != digest:
            raise RuntimeError("current native source differs from received H6 bytes: " + name)
        current_native_inputs[str(path)] = digest
    library = binding["library"]
    if Path(args.native_library) != Path(library["path"]) or args.native_sha256 != library["sha256"]:
        raise RuntimeError("fresh Product requires the exact received H6 native library")
    tracked = dict(SEED_SOURCE_RECEIVED["input_files"])
    tracked.update(recovery_inputs)
    tracked.update(inputs)
    tracked.update(current_native_inputs)
    tracked.update({str(root_path): binding["root_sha256"], str(original_path): binding["original_sha256"],
                    str(snapshot_path): snapshot_record["sha256"],
                    library["path"]: library["sha256"],
                    binding["member_child"]["path"]: binding["member_child"]["sha256"]})
    return snapshot, {"source_commit": CACHE_COMMIT, "native_source_commit": NATIVE_SOURCE_COMMIT,
        "native_repository": NATIVE_REPOSITORY, "native_sources_unchanged": 64,
        "native_recompiled": False, "qualification_reused": False,
        "source_receiving": SOURCE_ADMISSION, "native_admission": binding,
        "native_sources": native_sources, "library": library, "member_child": binding["member_child"],
        "input_files": tracked}


def receive_seed_package_inputs():
    """Receive a closed passive seed; no UV, namespace, native or Ready authority."""
    received = pinned_json(SEED_PACKAGE / "root-receiving.json", SEED_PACKAGE_ROOT_SHA)
    original = pinned_json(SEED_PACKAGE / "receipt.json", SEED_PACKAGE_RECEIPT_SHA, 4 << 20)
    if (received.get("schema") != "root-uv-cache-seed-package-receiving-v1"
            or received.get("status") != "RECEIVED_PASS" or received.get("receipt_sha256") != SEED_PACKAGE_RECEIPT_SHA
            or received.get("original_handles_closed") is not True
            or received.get("source_and_inputs_unchanged") is not True
            or received.get("archive_independently_verified") is not True
            or received.get("source_commit") != SEED_PARENT_COMMIT
            or received.get("parent", {}).get("pid") != 25776 or received["parent"].get("exit_code") != 0
            or received["parent"].get("status") != "COMPLETE" or received["parent"].get("terminal_exit_received") is not True
            or original.get("schema") != "uv-cache-seed-package-run-v1" or original.get("status") != "PASS"
            or original.get("run") != str(SEED_PACKAGE) or original.get("original_handles_closed") is not True
            or original.get("source_unchanged") is not True or original.get("archive_independently_verified") is not True
            or any(received.get(key) != row or original.get(key) != row for key, row in SEED_FILES.items())):
        raise RuntimeError("original closed seed package differs")
    inputs = {str(SEED_PACKAGE / "root-receiving.json"): SEED_PACKAGE_ROOT_SHA,
              str(SEED_PACKAGE / "receipt.json"): SEED_PACKAGE_RECEIPT_SHA}
    for row in SEED_FILES.values():
        path = ordinary(SEED_PACKAGE / row["path"], False)
        before = path.stat()
        measured, count = hashlib.sha256(), 0
        with path.open("rb") as reader:
            opened = os.fstat(reader.fileno())
            if not os.path.samestat(before, opened) or opened.st_size != row["bytes"]:
                raise RuntimeError("seed file identity/size differs")
            while True:
                block = reader.read(min(1 << 20, row["bytes"] - count + 1))
                if not block:
                    break
                count += len(block)
                if count > row["bytes"]:
                    raise RuntimeError("seed file exceeded its exact bound")
                measured.update(block)
            final = os.fstat(reader.fileno())
        after = path.stat()
        if (not os.path.samestat(opened, final) or not os.path.samestat(final, after)
                or (before.st_size, before.st_mtime_ns) != (final.st_size, final.st_mtime_ns)
                or (final.st_size, final.st_mtime_ns) != (after.st_size, after.st_mtime_ns)
                or count != row["bytes"] or measured.hexdigest() != row["sha256"]):
            raise RuntimeError("original seed file changed while receiving")
        inputs[str(path)] = row["sha256"]
    descriptor = pinned_json(SEED_PACKAGE / "seed-descriptor.json", SEED_FILES["descriptor"]["sha256"], 64 << 10)
    if (descriptor.get("schema") != "private-serena-uv-cache-seed-v1"
            or descriptor.get("archive") != SEED_FILES["archive"]
            or any(descriptor.get("manifest", {}).get(k) != v for k, v in SEED_FILES["manifest"].items())):
        raise RuntimeError("descriptor does not bind the exact fixed sibling seed files")
    return {"path": str(SEED_PACKAGE / "seed-descriptor.json"),
            "sha256": SEED_FILES["descriptor"]["sha256"]}, inputs


def build(args):
    admitted_source, native_reuse = receive_native_reuse(args)
    # The exact unchanged H6 native map is received before fresh Product creates its own isolated caches.
    if os.path.lexists(API_SYSO):
        raise RuntimeError("API compile-only must not consume a retained qualified source resource")
    if len(args.name) > 70:
        raise ValueError("build name must leave room for the qualified- artifact prefix")
    if not re.fullmatch(r"[0-9a-f]{40}", args.commit) or not re.fullmatch(r"[0-9a-f]{64}", args.native_sha256):
        raise ValueError("canonical source commit and native SHA-256 required")
    library = ordinary(Path(args.native_library), False)
    if not library.is_relative_to(Path(r"<TASK_WORKSPACE>")) or sha(library) != args.native_sha256:
        raise RuntimeError("explicit D-only native input differs from qualified SHA-256")
    for syso in (PRODUCT_SYSO, CLI_SYSO):
        if syso.exists():
            raise RuntimeError("pre-existing resource must not be overwritten: "+str(syso))
    proxy_admission = receive_module_proxy()
    if tree(MODULE_PROXY) != proxy_admission["manifest"]:
        raise RuntimeError("accepted imported module archive bytes changed before the fresh bootstrap")
    root = private_run(args.name)
    run = Run(root, clean_environment(root))
    staged_cli_hash = None
    try:
        run.receipt["scope_helper_sha256"] = HELPER_SHA
        run.receipt["input_files"] = {str(p):sha(ordinary(p, False)) for p in (GO,GIT,CC,PWSH,PYTHON,REPO/"build.ps1",Path(__file__),TOOL_RECEIPT,library,_guard_path)}
        host_receipt = pinned_json(WINPS_HOST_ADMISSION, WINPS_HOST_ADMISSION_SHA256)
        if (host_receipt.get("schema") != "native-winps-root-receiving-v1" or host_receipt.get("status") != "PASS"
                or host_receipt.get("commands_quiescent") is not True or host_receipt.get("guard_use_admitted") is not True
                or host_receipt.get("guard_sha256") != _guard_sha or host_receipt.get("powershell_path") != str(WINPS)
                or host_receipt.get("powershell_sha256") != WINPS_SHA256
                or sha(ordinary(WINPS, False)) != WINPS_SHA256):
            raise RuntimeError("original native Windows PowerShell host admission differs")
        run.receipt["input_files"].update({str(WINPS): WINPS_SHA256,
            str(WINPS_HOST_ADMISSION): WINPS_HOST_ADMISSION_SHA256, **NATIVE_UTILITY_PINS})
        if any(sha(ordinary(path, False)) != digest for path, digest in NATIVE_UTILITY_PINS.items()):
            raise RuntimeError("pinned native Windows PowerShell Utility inputs differ")
        run.receipt["product_host"] = {"path": str(WINPS), "sha256": WINPS_SHA256,
            "qualification": {"path": str(WINPS_HOST_ADMISSION), "sha256": WINPS_HOST_ADMISSION_SHA256},
            "utility_inputs": NATIVE_UTILITY_PINS}
        run.receipt["input_files"].update(native_reuse["input_files"])
        run.receipt["input_files"].update(CACHE_HANDOFF_INPUTS)
        run.receipt["input_files"].update({str(MODULE_PROXY / name): row["sha256"]
            for name, row in proxy_admission["manifest"].items()})
        run.receipt["native_reuse"] = native_reuse
        run.receipt["source_before"] = product_source_snapshot(run, "before")
        if run.receipt["source_before"] != admitted_source or admitted_source["head"] != args.commit:
            raise RuntimeError("actual source differs from its accepted Ready/finalization commit")
        proxy, proxy_record = resource_tool_proxy(root)
        run.env["GOPROXY"] = proxy.as_uri()+","+run.env["GOPROXY"]
        run.receipt["environment"]["GOPROXY"] = run.env["GOPROXY"]
        run.receipt["resource_tool_proxy"] = proxy_record
        run.receipt["input_files"].update(proxy_record["source_files"])
        run.save()
        artifact = REPO/".scratch"/("qualified-"+args.name)
        artifact_security = scope_helper().create_private_artifact_scope(artifact)
        new_json(root/"artifact-scope-security.json", artifact_security)
        new_json(root/"cleanup-contract.json", {"artifact":str(artifact),"runner_sha256":sha(__file__)})
        wrapper = builder_script(run, artifact, library)
        run.command([WINPS,"-NoLogo","-NoProfile","-File",wrapper], "actual-builder", timeout=900)
        complete_product_cleanup(run, artifact)
        if PRODUCT_SYSO.exists():
            raise RuntimeError("actual builder left its source resource in place")
        plan = json.loads((root/"actual-build-plan.json").read_text(encoding="utf-8"))
        if plan["commit"] != args.commit or not plan["testOnly"] or plan["buildTags"] != ["test_state_path_env"] or len(plan["artifacts"]) != 2:
            raise RuntimeError("actual test-only build plan changed")
        if any(a["buildTags"] != ["test_state_path_env"] for a in plan["artifacts"]):
            raise RuntimeError("test-only isolation is not paired")
        actual = root/"product"
        actual.mkdir()
        for rel in BUNDLE_FILES:
            copy_file(artifact/Path(rel), actual/Path(rel))
        actual_record = bundle_admission(run, actual, plan, "actual", args.native_sha256)
        preserved_resources = []
        for record in sorted((root/"cleanup-records").glob("*.json")):
            row = json.loads(record.read_text(encoding="utf-8"))
            if Path(row["source"]) == PRODUCT_SYSO:
                preserved_resources.append(Path(row["preserved"]))
        if len(preserved_resources) != 1:
            raise RuntimeError("one actual Product resource must have been preserved before trash")
        resource = ordinary(preserved_resources[0], False)
        resource_copy = root/"resources"/"product-versioninfo.syso"
        copy_file(resource, resource_copy)
        cli = root/"cli-test-product"
        cli.mkdir()
        for rel in BUNDLE_FILES:
            if rel not in ("mcphub-product/mcphub.exe","windows-product-identity.json"):
                copy_file(actual/Path(rel), cli/Path(rel))
        staged_cli_hash = copy_file(resource_copy, CLI_SYSO)
        ldflags = " ".join(["-H=windowsgui","-X mcp-local-hub/internal/buildinfo.version="+plan["version"],
            "-X mcp-local-hub/internal/buildinfo.commit="+plan["commit"],
            "-X mcp-local-hub/internal/buildinfo.date="+plan["buildDate"]])
        try:
            run.command([GO,"test","-c","-trimpath","-tags","test_state_path_env","-ldflags",ldflags,
                         "-o",cli/"mcphub-product"/"mcphub.exe","./internal/cli"], "cli-test-compile", timeout=600)
        finally:
            if run.commands_quiescent():
                if CLI_SYSO.exists():
                    if sha(CLI_SYSO) != staged_cli_hash:
                        raise RuntimeError("staged CLI resource changed; retained for review")
                    cleanup_move(root, CLI_SYSO, owner_run=run)
            else:
                run.receipt["retained_source_resource"] = {
                    "path": str(CLI_SYSO), "expected_sha256": staged_cli_hash,
                    "reason": "original compiler completion is unconfirmed; no observation or disposition"}
        product_identity = run.validator(["--candidate",str(cli/"mcphub-product"/"mcphub.exe"),
            "--expected-version",plan["version"],"--expected-commit",plan["commit"],"--expected-build-date",plan["buildDate"]], "cli-product-identity")
        new_json(cli/"windows-product-identity.json", product_identity)
        cli_record = bundle_admission(run, cli, plan, "cli", args.native_sha256)
        actual_record["product_rust_marked"] = True
        cli_record["product_rust_marked"] = False
        cli_record["test_build_tags"] = ["test_state_path_env"]
        api_image = root / "api-compile-only.test.exe"
        if os.path.lexists(api_image):
            raise RuntimeError("API compilation requires one new output in the Product run")
        run.command([GO, "test", "-c", "-trimpath", "-tags", "test_state_path_env",
                     "-o", api_image, "./internal/api"], "api-test-compile-only", timeout=600)
        run.receipt["api_test_compilation"] = {"path": str(api_image), "sha256": sha(ordinary(api_image, False)),
            "build_tags": ["test_state_path_env"], "tests_executed": False,
            "qualification_passed": False, "command": "api-test-compile-only"}
        run.receipt.update({"actual_product":actual_record,"cli_test_product":cli_record,"build_plan":plan,
            "resource_sha256":sha(resource_copy),"native_input_sha256":args.native_sha256,
            "qualification":"compiled artifacts plus portable and native entry-set/sidecar admission; no launcher execution, tests or console qualification"})
        run.receipt["artifact_staging_disposition"] = cleanup_move(root, artifact, owner_run=run)
        after = product_source_snapshot(run, "after")
        if after != run.receipt["source_before"] or PRODUCT_SYSO.exists() or CLI_SYSO.exists() or os.path.lexists(API_SYSO):
            raise RuntimeError("source tree changed during coordinated fixture build")
        run.receipt["source_unchanged"] = True
        if any(sha(path) != digest for path, digest in run.receipt["input_files"].items()):
            raise RuntimeError("original build tool or native input changed")
        run.receipt["original_inputs_unchanged"] = True
        run.receipt["commands_quiescent"] = run.commands_quiescent()
        run.receipt["command_quiescence"] = run.quiescence_snapshot()
        if tree(proxy) != proxy_record["files"]:
            raise RuntimeError("per-run resource-tool proxy changed")
        run.receipt["resource_tool_proxy_unchanged"] = True
        new_json(root/"next-qualification.json", {
            "build_run": str(root),
            "status": "compiled and admitted only; no runtime execution",
            "remaining": ["actual launcher receiving", "fixed canonical GUI replay diagnostic"],
            "execution_rule": "use a separately reviewed original-owner receiver; this runner admits only build and owned cleanup"})
        run.finish("PASS")
    except BaseException as exc:
        if run.commands_quiescent():
            try:
                if "artifact" in locals() and "deferred_cleanup" not in run.receipt and (root / "deferred-cleanup.json").exists():
                    complete_product_cleanup(run, artifact)
                if "artifact" in locals() and artifact.exists():
                    run.receipt["artifact_staging_disposition"] = cleanup_move(root, artifact, owner_run=run)
                after_failure = product_source_snapshot(run, "after-failure")
                expected_source = run.receipt.get("source_before", locals().get("before"))
                run.receipt["pending_source_resources"] = [str(p) for p in (PRODUCT_SYSO, CLI_SYSO) if p.exists()]
                run.receipt["source_unchanged"] = expected_source is not None and after_failure == expected_source and not run.receipt["pending_source_resources"]
            except BaseException as observation_error:
                run.receipt["source_observation_error"] = type(observation_error).__name__+": "+str(observation_error)
        else:
            run.receipt["finalization_observations"] = {
                "source_after": "NOT_PERFORMED", "resource_disposition": "NOT_PERFORMED",
                "input_hashes": "NOT_PERFORMED",
                "reason": "original command completion is unconfirmed"}
        run.receipt["commands_quiescent"] = run.commands_quiescent()
        run.receipt["command_quiescence"] = run.quiescence_snapshot()
        run.finish("FAIL", exc)
        raise

def received_bundle(args, which):
    donor = receive_product_donor(Path(args.build_run))
    receipt = donor["value"]
    if which not in ("actual_product", "cli_test_product"):
        raise RuntimeError("only the two independently admitted Product bundles are available")
    bundle = receipt[which]
    if tree(Path(bundle["path"])) != bundle["files"]:
        raise RuntimeError("compiled bundle changed since independent artifact admission")
    return donor["root"], receipt, bundle

def execute(args, cli):
    original, built, bundle = received_bundle(args, "cli_test_product" if cli else "actual_product")
    root = private_run(args.name)
    env = clean_environment(root)
    if cli:
        env.update({"MCPHUB_CANONICAL_ENTRY_FIXTURE":bundle["fixture"],"MCPHUB_CANONICAL_FIXTURE_TEST_PRODUCT":"1",
                    "MCPHUB_CANONICAL_ENTRY_FIXTURE_REQUIRED":"1","MCPHUB_CANONICAL_COMMAND_FIXTURE_REQUIRED":"1"})
    run = Run(root, env)
    try:
        run.receipt.update({"build_run":str(original),"bundle":bundle,"scope_helper_sha256":HELPER_SHA})
        before = source_snapshot(run, "before")
        if before != built["source_before"]:
            raise RuntimeError("tests or probe must receive the exact built source snapshot")
        if cli:
            exe = Path(bundle["path"])/"mcphub-product"/"mcphub.exe"
            run.command([exe,"-test.v=test2json","-test.count=1","-test.timeout=120s","-test.run="+args.pattern], "qualified-cli", timeout=150, check=False)
            test_exit = run.receipt["commands"][-1]["exit_code"]
            raw = root/"qualified-cli.stdout.txt"
            converted = run.command([GO,"tool","test2json","-p","mcp-local-hub/internal/cli"], "test2json", timeout=30, stdin_path=raw)
            events = [json.loads(line) for line in converted.splitlines()]
            selected = [e for e in events if e.get("Test") and e.get("Action") in ("pass","fail","skip")]
            run.receipt["test_events"] = selected
            if test_exit or any(e["Action"] == "fail" for e in selected):
                raise RuntimeError("qualified CLI tests failed; exact outcomes are retained")
            if not any(e["Action"] == "pass" for e in selected):
                raise RuntimeError("selected qualified CLI tests produced no PASS")
        else:
            output = run.command([Path(bundle["path"])/"mcphub.exe","version"], "actual-launcher-version", timeout=30)
            plan = built["build_plan"]
            expected_product = str(Path(bundle["path"])/"mcphub-product"/"mcphub.exe")
            reported_paths = [line.split("running:", 1)[1].strip() for line in output.splitlines() if line.strip().startswith("running:")]
            if "mcp-local-hub "+plan["version"] not in output or "commit:     "+plan["commit"] not in output or len(reported_paths) != 1 or not os.path.samefile(reported_paths[0], expected_product):
                raise RuntimeError("actual launcher did not report its qualified Product")
            run.receipt["qualification"] = "actual launcher -> actual Product version with regular redirected stdio; native held entry-set/fence/borrow path required by the compiled receiver; no console/ConPTY qualification"
        if source_snapshot(run, "after") != before or tree(Path(bundle["path"])) != bundle["files"]:
            raise RuntimeError("source or original qualified bundle changed during execution")
        run.receipt["source_unchanged"] = True
        run.receipt["bundle_unchanged"] = True
        run.finish("PASS")
    except BaseException as exc:
        if run.commands_quiescent():
            try:
                after_failure = source_snapshot(run, "after-failure")
                expected_source = run.receipt.get("source_before", locals().get("before"))
                run.receipt["pending_source_resources"] = [str(p) for p in (PRODUCT_SYSO, CLI_SYSO) if p.exists()]
                run.receipt["source_unchanged"] = expected_source is not None and after_failure == expected_source and not run.receipt["pending_source_resources"]
            except BaseException as observation_error:
                run.receipt["source_observation_error"] = type(observation_error).__name__+": "+str(observation_error)
        else:
            run.receipt["finalization_observations"] = {
                "source_after": "NOT_PERFORMED", "resource_disposition": "NOT_PERFORMED",
                "input_hashes": "NOT_PERFORMED",
                "reason": "original command completion is unconfirmed"}
        run.receipt["commands_quiescent"] = run.commands_quiescent()
        run.receipt["command_quiescence"] = run.quiescence_snapshot()
        run.finish("FAIL", exc)
        raise

def main():
    parser = argparse.ArgumentParser(description=__doc__)
    sub = parser.add_subparsers(dest="action", required=True)
    p = sub.add_parser("build")
    p.add_argument("--name", required=True, type=scope_name)
    p.add_argument("--native-library", required=True)
    p.add_argument("--native-sha256", required=True)
    p.add_argument("--commit", required=True)
    for action in ("cli-test","launcher-probe"):
        p = sub.add_parser(action)
        p.add_argument("--name", required=True, type=scope_name)
        p.add_argument("--build-run", required=True)
        if action == "cli-test":
            p.add_argument("--pattern", required=True)
    p = sub.add_parser("cleanup")
    p.add_argument("--run", required=True)
    p.add_argument("--path", required=True)
    p.add_argument("--builder-pid", type=int)
    args = parser.parse_args()
    if args.action == "build":
        build(args)
    elif args.action == "cleanup":
        raise RuntimeError("v6 cleanup is in-process after the retained original Run closes")
    else:
        raise RuntimeError("this versioned runner is build/owned-cleanup only; execution requires an original owner receiver")

if __name__ == "__main__":
    main()
