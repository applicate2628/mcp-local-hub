def require(condition, reason):
    if not condition:
        raise RuntimeError(reason)


def ordinary(path):
    path = Path(path)
    for parent in [*reversed(path.parents), path]:
        info = parent.lstat()
        require(not stat.S_ISLNK(info.st_mode) and not getattr(info, "st_file_attributes", 0) & 0x400,
                "redirected DATA input: " + str(parent))
    require(path.is_file(), "ordinary DATA file required: " + str(path))
    return path


def pinned(path, expected, limit=16 << 20):
    require(REVIEW_COMPLETE is True and isinstance(expected, str) and re.fullmatch(r"[0-9a-f]{64}", expected),
            "fresh full inputs remain disabled or an actual pin is missing")
    with ordinary(path).open("rb") as stream:
        raw = stream.read(limit + 1)
    require(len(raw) <= limit and hashlib.sha256(raw).hexdigest() == expected,
            "pinned fresh full input changed: " + str(path))
    return raw
