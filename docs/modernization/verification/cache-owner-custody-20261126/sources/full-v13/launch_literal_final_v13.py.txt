def pinned_bytes(path, expected, limit=16 << 20):
    if not isinstance(expected, str) or re.fullmatch(r"[0-9a-f]{64}", expected) is None:
        raise RuntimeError("actual input pin required: " + str(path))
    path = Path(path)
    for part in (*reversed(path.parents), path):
        info = part.lstat()
        if stat.S_ISLNK(info.st_mode) or getattr(info, "st_file_attributes", 0) & 0x400:
            raise RuntimeError("ordinary input path required: " + str(part))
    if not path.is_file():
        raise RuntimeError("ordinary input file required: " + str(path))
    with path.open("rb") as stream:
        raw = stream.read(limit + 1)
    if len(raw) > limit or hashlib.sha256(raw).hexdigest() != expected:
        raise RuntimeError("bounded pinned input differs: " + str(path))
    return raw

            finally:
                # Keep the original Popen object and both original capture
                # handles until direct exit; no PID reconstruction or kill.
                while code is None:
                    try:
                        code = original.wait()
                    except BaseException as error:
                        # An interrupted wait does not release our original
                        # process or its captures and is never an exit receipt.
                        receipt["launcher_error"] = type(error).__name__ + ": " + str(error)
                elapsed = time.monotonic() - started
                receipt.update(state="direct-reaped", exit_code=code, seconds=elapsed,
                               terminal_exit_received=True,
                               deadline_exceeded=receipt["deadline_exceeded"] or elapsed > OUTER_DIAGNOSTIC_SECONDS)
    except BaseException as error:
        receipt["launcher_error"] = type(error).__name__ + ": " + str(error)
    receipt["parent_capture_files_closed"] = bool(out is not None and err is not None and out.closed and err.closed)
    if receipt["terminal_exit_received"] is not True:
        receipt["state"] = "direct-completion-unreceived"
        save()
        print(json.dumps(receipt), flush=True)
        return 1
    save()
    report_path = ISSUER_ROOT / "receipt.json"
    if report_path.is_file():
        try:
            with report_path.open("rb") as stream:
                raw = stream.read((16 << 20) + 1)
            if len(raw) > 16 << 20:
                raise RuntimeError("literal driver control exceeds its declared bound")
            report = json.loads(raw)
            receipt.update(child_status=report.get("status"), commands_quiescent=report.get("commands_quiescent"),
                           command_count=len(report.get("commands", [])), failure=report.get("failure"),
                           candidate_commit=report.get("source_before", {}).get("head"), child_schema=report.get("schema"),
                           literal_commands=report.get("literal_commands"),
                           product_test_source_compatibility=report.get("product_test_source_compatibility"),
                           literal_pre_push_commands_passed=report.get("literal_pre_push_commands_passed"),
                           original_receipt_sha256=hashlib.sha256(raw).hexdigest())
        except BaseException as error:
            receipt.update(child_status="UNREADABLE_REPORT", launcher_error=type(error).__name__ + ": " + str(error))
    else:
        receipt.update(child_status="NO_REPORT", failure="issuer exited before writing its fixed receipt")
    receipt["state"] = "received"
    save()
    passed = (code == 0 and receipt["deadline_exceeded"] is False and "launcher_error" not in receipt
              and receipt["parent_capture_files_closed"] is True and receipt.get("child_status") == "PASS"
              and receipt.get("commands_quiescent") is True and receipt.get("candidate_commit") == SOURCE_COMMIT
              and receipt.get("child_schema") == "literal-go-prepush-gate-v1"
              and receipt.get("literal_commands") == EXPECTED_LITERAL_COMMANDS
              and receipt.get("literal_pre_push_commands_passed") is True
              and isinstance(receipt.get("product_test_source_compatibility"), dict)
              and receipt["product_test_source_compatibility"].get("product_source_commit") == PRODUCT_SOURCE_COMMIT
              and receipt["product_test_source_compatibility"].get("test_source_commit") == SOURCE_COMMIT
              and receipt["product_test_source_compatibility"].get("changed_file_count") == 11
              and receipt["product_test_source_compatibility"].get("unchanged_non_test_paths") is True)
    print(json.dumps({"parent": receipt, "parent_receipt": str(PARENT_RECEIPT),
                      "parent_receipt_sha256": hashlib.sha256(PARENT_RECEIPT.read_bytes()).hexdigest()}), flush=True)
    return 0 if passed else 1
