def require(value, message):
    if not value:
        raise RuntimeError(message)


def ordinary(path, directory):
    path = Path(path)
    for parent in reversed(path.parents):
        info = parent.lstat()
        require(stat.S_ISDIR(info.st_mode) and not info.st_file_attributes & 0x400,
                "nonordinary ancestor: " + str(parent))
    info = path.lstat()
    require(not info.st_file_attributes & 0x400
            and (stat.S_ISDIR(info.st_mode) if directory else stat.S_ISREG(info.st_mode)),
            "nonordinary path: " + str(path))
    return info


def pinned(pair):
    path = Path(pair["path"])
    before = ordinary(path, False)
    require(before.st_size <= LIMIT, "metadata input exceeds bound")
    with path.open("rb") as stream:
        opened = os.fstat(stream.fileno())
        require((opened.st_dev, opened.st_ino) == (before.st_dev, before.st_ino),
                "metadata object changed during open")
        raw = stream.read(LIMIT + 1)
    require(len(raw) <= LIMIT and hashlib.sha256(raw).hexdigest() == pair["sha256"],
            "exact metadata digest differs: " + str(path))
    return raw

def open_locked_directory(scope, path, for_rename=False):
    # LIST_DIRECTORY makes the no-delete-sharing lock effective for the name.
    # DELETE access authorizes rename of this original handle; it does not delete.
    access = scope.READ_CONTROL | scope.FILE_READ_ATTRIBUTES | 0x1 | 0x20
    if for_rename:
        access |= 0x00010000
    handle = scope.K.CreateFileW(str(path), access, 1 | 2, None, 3, 0x02200000, None)
    return scope.handle_scope(handle)


def identity(scope, handle, path):
    native = scope.directory_identity(handle)
    info = ordinary(path, True)
    require(native["file_index"] == info.st_ino and native["volume_serial"] == info.st_dev & 0xffffffff,
            "path and retained native handle identify different objects")
    return {"native": native, "stat": {"device": info.st_dev, "inode": info.st_ino}}


def require_empty(path):
    with os.scandir(path) as entries:
        require(next(entries, None) is None, "directory must be empty: " + str(path))


def rename_original(scope, handle, destination):
    class RenameInfo(C.Structure):
        _fields_ = [("flags", C.c_uint32), ("root", C.c_void_p),
                    ("length", C.c_uint32), ("name", C.c_uint16 * 1)]
    if C.sizeof(C.c_void_p) != 8 or RenameInfo.name.offset != 20:
        raise RuntimeError("unexpected FILE_RENAME_INFO layout")
    encoded = str(destination).encode("utf-16-le")
    raw = C.create_string_buffer(max(C.sizeof(RenameInfo), RenameInfo.name.offset + len(encoded) + 2))
    value = RenameInfo.from_buffer(raw)
    value.root, value.length = None, len(encoded)
    C.memmove(C.addressof(raw) + RenameInfo.name.offset, encoded, len(encoded))
    function = scope.bind(scope.K, "SetFileInformationByHandle", C.c_int,
                          [C.c_void_p, C.c_int, C.c_void_p, C.c_uint32])
    scope.require(function(handle, 3, raw, len(raw)), "absolute original-directory rename")


def append_record(stream, value):
    raw = (json.dumps(value, ensure_ascii=False, separators=(",", ":")) + "\n").encode("utf-8")
    stream.write(raw)
    stream.flush()
    os.fsync(stream.fileno())


def retire(plan, scope):
    bucket, evidence = Path(plan["trash_bucket"]), Path(plan["evidence_root"])
    sources = [Path(row["source"]) for row in plan["resources"]]
    require(bucket == TASK / "trash" / "closed-full-v9-20261126-v2"
            and evidence == TASK / "closed-v9-retirement-v2", "fixed authorized destination paths required")
    with contextlib.ExitStack() as retained:
        parents = set()
        for path in sources + [bucket / "placeholder", evidence / "placeholder"]:
            parents.update(path.parents)
        parent_handles = {}
        for path in sorted(parents, key=lambda p: (len(p.parts), str(p))):
            ordinary(path, True)
            handle = retained.enter_context(open_locked_directory(scope, path))
            scope.directory_identity(handle)
            parent_handles[path] = handle
        require_empty(bucket)
        require_empty(evidence)
        bucket_identity = identity(scope, parent_handles[bucket], bucket)
        evidence_identity = identity(scope, parent_handles[evidence], evidence)
        checked = []
        for row, source in zip(plan["resources"], sources):
            ordinary(source, True)
            handle = retained.enter_context(open_locked_directory(scope, source, True))
            before = identity(scope, handle, source)
            require(before["native"] == row["native_identity"], "original native resource identity changed")
            if row["kind"] == "cache":
                require(before["stat"] == row["creation_identity"], "original cache creation identity changed")
            require(before["stat"]["device"] == bucket_identity["stat"]["device"], "cross-volume move refused")
            if row["require_empty"]:
                require_empty(source)
            destination = bucket / row["name"]
            require(not os.path.lexists(destination), "existing trash target refused")
            checked.append((row, source, destination, handle, before))
        # All four objects are checked and held before the first move.
        summary = {"schema": "closed-v9-four-resource-retirement-v1", "status": "IN_PROGRESS",
                   "plan_sha256": PLAN_SHA256, "controls": plan["controls"],
                   "bucket_identity": bucket_identity, "evidence_identity": evidence_identity,
                   "moved": [], "delete_copy_replace_or_acl_change": False,
                   "raw_captures_or_cache_contents_read": False, "original_failure_rewritten": False}
        try:
            for row, source, destination, handle, before in checked:
                journal = evidence / (row["name"] + ".jsonl")
                with journal.open("xb") as stream:
                    base = {"schema": "original-directory-trash-move-v1", "name": row["name"],
                            "source": str(source), "destination": str(destination),
                            "creation_identity": row["creation_identity"], "before": before,
                            "plan_sha256": PLAN_SHA256, "empty_required": row["require_empty"]}
                    append_record(stream, dict(base, phase="PREPARED"))
                    try:
                        require(identity(scope, handle, source) == before, "source changed before rename")
                        if row["require_empty"]:
                            require_empty(source)
                        rename_original(scope, handle, destination)
                        after = identity(scope, handle, destination)
                        require(after == before and not os.path.lexists(source), "rename identity postcondition differs")
                        if row["require_empty"]:
                            require_empty(destination)
                        append_record(stream, dict(base, phase="MOVED", after=after, source_absent=True))
                        summary["moved"].append({"name": row["name"], "destination": str(destination),
                                                 "identity": after, "journal": str(journal)})
                    except BaseException as error:
                        append_record(stream, dict(base, phase="ERROR", error=repr(error),
                                                   automatic_retry_or_rollback=False))
                        raise
            require(identity(scope, parent_handles[bucket], bucket) == bucket_identity
                    and identity(scope, parent_handles[evidence], evidence) == evidence_identity,
                    "original output parent identity changed")
            summary["status"] = "PASS"
        except BaseException as error:
            summary["status"], summary["error"] = "FAIL", repr(error)
            raise
        finally:
            with (evidence / "receipt.json").open("xb") as stream:
                append_record(stream, summary)
    return summary


def main():
    require(REVIEW_COMPLETE is True, "disabled: exact root review required before upload/execution")
    require(os.name == "nt" and not sys.argv[1:] and sys.flags.isolated and sys.flags.no_site
            and sys.flags.dont_write_bytecode and Path.cwd() == TASK and Path(__file__).absolute() == SELF,
            "exact native Python -I -S -B, path, cwd and no arguments required")
    plan = json.loads(pinned({"path": str(PLAN), "sha256": PLAN_SHA256}))
    require(plan.get("schema") == "closed-v9-four-resource-retirement-plan-v1"
            and plan.get("task") == str(TASK) and plan.get("issuer") == str(SELF), "exact task-bound plan required")
    require(Path(sys.executable).resolve(strict=True) == Path(plan["python"]["path"]), "native Python path differs")
    pinned(plan["python"])
    values = {name: json.loads(pinned(pair)) for name, pair in plan["controls"].items()}
    receive_controls(plan, values)
    code = pinned(plan["identity_helper"])
    scope = types.ModuleType("retirement_identity_helper")
    scope.__file__ = plan["identity_helper"]["path"]
    exec(compile(code, scope.__file__, "exec"), scope.__dict__)
    result = retire(plan, scope)
    print(json.dumps({"status": result["status"], "moved": len(result["moved"]),
                      "receipt": str(Path(plan["evidence_root"]) / "receipt.json")}), flush=True)
    return 0
