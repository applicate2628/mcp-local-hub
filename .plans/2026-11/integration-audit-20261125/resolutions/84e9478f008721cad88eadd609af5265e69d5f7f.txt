func TestCLITestBinaryRejectsMalformedHelperDispatch(t *testing.T) {
	tests := []struct {
		name string
		args []string
		env  []string
	}{
		{name: "exit sentinel without selector", args: []string{"-test.run=^$"}, env: []string{"MCPHUB_TEST_CHILD_EXIT_CODE=42"}},
		{name: "pid sentinel with wrong selector", args: []string{"-test.run=^TestDoesNotExist$"}, env: []string{"MCPHUB_PID_MATCH_HELPER=1"}},
		{name: "selector without sentinel", args: []string{"-test.run=^TestProductionTerminateFn_HelperSleep$"}},
		{name: "conflicting helpers", args: []string{"-test.run=^$"}, env: []string{"MCPHUB_PRODUCTION_TERMINATE_HELPER=1", "MCPHUB_PID_MATCH_HELPER=1"}},
		{name: "unknown overlay sentinel value", args: []string{"-test.run=^$"}, env: []string{overlayMarkerHelperSentinelEnv + "=2"}},
		{name: "raw route", args: []string{"route"}},
		{name: "incomplete route", args: []string{"route", "--port"}},
		{name: "raw supervise", args: []string{"supervise"}},
	}

	for _, tc := range tests {
		t.Run(tc.name, func(t *testing.T) {
			cmd := exec.Command(os.Args[0], tc.args...)
			process.NoConsole(cmd)
			cmd.Env = append(withoutCLITestHelperEnvironment(os.Environ()), tc.env...)
			var stderr bytes.Buffer
			cmd.Stderr = &stderr
			if err := cmd.Run(); err == nil {
				t.Fatalf("malformed helper dispatch exited 0; stderr=%q", stderr.String())
			}
		})
	}
}

func withoutCLITestHelperEnvironment(env []string) []string {
	keys := []string{
		"MCPHUB_TEST_CHILD_EXIT_CODE",
		staleExitHelperSentinelEnv,
		staleExitHelperReleaseEnv,
		"MCPHUB_PRODUCTION_TERMINATE_HELPER",
		"MCPHUB_PID_MATCH_HELPER",
		overlayMarkerHelperSentinelEnv,
		overlayMarkerHelperDumpPathEnv,
		forensicsSinkChildEnv,
		forensicsSinkChildModeEnv,
		cliProductionArgvHelperEnv,
		concurrentSecretsRootEnv,
		"MCPHUB_CODEGRAPH_COMPAT_HELPER",
		"MCPHUB_MAPPED_IMAGE_RENAME_HELPER",
		"MCPHUB_TEST_OWNED_PRECONTROL_CHILD",
		"MCPHUB_DAEMON_ROOT_EXIT_CHILD_CODE",
	}
	keys = append(keys, obsoleteConcurrentSecretsHelperEnv[:]...)
	result := make([]string, 0, len(env))
	for _, entry := range env {
		name, _, _ := strings.Cut(entry, "=")
		keep := true
		for _, key := range keys {
			if strings.EqualFold(name, key) {
				keep = false
				break
			}
		}
		if keep {
			result = append(result, entry)
		}
	}
	return result
}

func TestCLITestHelperDispatchConcurrentSecretsRole(t *testing.T) {
	base, err := filepath.Abs(os.TempDir())
	if err != nil {
		t.Fatal("resolve fixture temp-base syntax")
	}
	root := filepath.Join(base, concurrentSecretsDirPrefix+"fixture", concurrentSecretsStateLeaf)
	selector := []string{"fixture-test-binary", concurrentSecretsSelector}
	validEnv := map[string]string{concurrentSecretsRootEnv: root, api.RequireSingleUserHomeEnv: "1"}
	for _, tc := range []struct {
		name           string
		args           []string
		env            map[string]string
		reject, child  bool
		ordinaryHelper bool
	}{
		{name: "same selector ordinary parent", args: selector},
		{name: "exact fixture child", args: selector, env: validEnv, child: true},
		{name: "existing separate role", args: []string{"fixture-test-binary", "-test.run=^TestFormatChildExit_RealProcessShowsExitCode$"},
			env: map[string]string{"MCPHUB_TEST_CHILD_EXIT_CODE": "42"}, ordinaryHelper: true},
		{name: "empty root is malformed", args: selector, env: map[string]string{concurrentSecretsRootEnv: "", api.RequireSingleUserHomeEnv: "1"}, reject: true},
		{name: "missing strict policy", args: selector, env: map[string]string{concurrentSecretsRootEnv: root}, reject: true},
		{name: "relative root", args: selector, env: map[string]string{concurrentSecretsRootEnv: concurrentSecretsStateLeaf, api.RequireSingleUserHomeEnv: "1"}, reject: true},
		{name: "non-fixture root", args: selector, env: map[string]string{concurrentSecretsRootEnv: filepath.Join(base, concurrentSecretsStateLeaf), api.RequireSingleUserHomeEnv: "1"}, reject: true},
		{name: "unanchored selector", args: []string{"fixture-test-binary", "-test.run=TestSecretsConcurrentSetHelperProcess"}, env: validEnv, reject: true},
		{name: "extra args", args: append(append([]string(nil), selector...), "-test.v"), env: validEnv, reject: true},
		{name: "conflicting role", args: selector, env: map[string]string{
			concurrentSecretsRootEnv: root, api.RequireSingleUserHomeEnv: "1", "MCPHUB_TEST_CHILD_EXIT_CODE": "42",
		}, reject: true},
		{name: "legacy partial envelope", args: selector, env: map[string]string{obsoleteConcurrentSecretsHelperEnv[2]: ""}, reject: true},
	} {
		t.Run(tc.name, func(t *testing.T) {
			dispatch := classifyCLITestHelperDispatch(tc.args, func(key string) (string, bool) {
				value, ok := tc.env[key]
				return value, ok
			}, nil)
			if dispatch.invalid != tc.reject || dispatch.runSelectedTest != (tc.child || tc.ordinaryHelper) ||
				(dispatch.concurrentSecretsRoot != "") != tc.child {
				t.Fatal("unexpected concurrent fixture admission")
			}
		})
	}
}

func TestWithoutCLITestHelperEnvironmentStripsConcurrentSecretsControls(t *testing.T) {
	env := []string{"PATH=fixture", strings.ToLower(concurrentSecretsRootEnv) + "=fixture", "MCPHUB_PID_MATCH_HELPER=1"}
	for _, key := range obsoleteConcurrentSecretsHelperEnv {
		env = append(env, key+"=fixture")
	}
	filtered := withoutCLITestHelperEnvironment(env)
	if len(filtered) != 1 || filtered[0] != "PATH=fixture" {
		t.Fatal("helper environment did not discard role controls")
	}
}

func cliTestStateRoots(t *testing.T) map[string]struct{} {
	t.Helper()
	paths, err := filepath.Glob(filepath.Join(os.TempDir(), "mcphub-cli-test-state-*"))
	if err != nil {
		t.Fatalf("glob CLI test-state roots: %v", err)
	}
	roots := make(map[string]struct{}, len(paths))
	for _, path := range paths {
		roots[path] = struct{}{}
	}
	return roots
}

func TestCLITestBinaryHelpers_BypassPackageStateRoot(t *testing.T) {
	tests := []struct {
		name    string
		args    []string
		env     []string
		kill    bool
		prepare func(*testing.T) []string
	}{
		{
			name: "exit-code",
			args: []string{"-test.run=^TestFormatChildExit_RealProcessShowsExitCode$"},
			env:  []string{"MCPHUB_TEST_CHILD_EXIT_CODE=42"},
		},
		{
			name: "stale-exit-release",
			args: []string{"-test.run=^TestStaleExitReleaseHelper$"},
			prepare: func(t *testing.T) []string {
				releasePath := filepath.Join(t.TempDir(), "release")
				if err := os.WriteFile(releasePath, []byte("release"), 0o600); err != nil {
					t.Fatalf("create stale-exit release marker: %v", err)
				}
				return []string{
					staleExitHelperSentinelEnv + "=1",
					staleExitHelperReleaseEnv + "=" + releasePath,
				}
			},
		},
		{
			name: "production-terminate",
			args: []string{"-test.run=^TestProductionTerminateFn_HelperSleep$"},
			env:  []string{"MCPHUB_PRODUCTION_TERMINATE_HELPER=1"},
			kill: true,
		},
		{
			name: "pid-match",
			args: []string{"-test.run=^TestPidMatchesMcphub_HelperSleep$"},
			env:  []string{"MCPHUB_PID_MATCH_HELPER=1"},
			kill: true,
		},
		{
			name: "production-supervisor-argv",
			args: []string{"supervise"},
			env:  []string{cliProductionArgvHelperEnv + "=1"},
		},
	}

	for _, tc := range tests {
		t.Run(tc.name, func(t *testing.T) {
			before := cliTestStateRoots(t)
			env := append([]string{}, tc.env...)
			if tc.prepare != nil {
				env = append(env, tc.prepare(t)...)
			}

			cmd := exec.Command(os.Args[0], tc.args...)
			process.NoConsole(cmd)
			cmd.Env = append(os.Environ(), env...)
			if err := cmd.Start(); err != nil {
				t.Fatalf("start helper child: %v", err)
			}
			if tc.kill {
				time.Sleep(100 * time.Millisecond)
				if err := cmd.Process.Kill(); err != nil {
					t.Fatalf("kill helper child: %v", err)
				}
			}
			_ = cmd.Wait()

			for path := range cliTestStateRoots(t) {
				if _, existed := before[path]; !existed {
					t.Errorf("helper child left package test-state root %s", path)
				}
			}
		})
	}
}
